## Build Your First Material App

This guide will walk you through building a simple Angular Material application from scratch. You'll learn how to import components, use them in your templates, and apply Material Design principles.

<Info>
  Make sure you've completed the [Installation](/installation) steps before starting this guide.
</Info>

## Step 1: Import Your First Component

Let's start by adding a Material slide toggle to your application.

<Steps>
  <Step title="Import the Module">
    In your component file, import `MatSlideToggleModule`:
    
    ```typescript app.component.ts
    import { Component } from '@angular/core';
    import { MatSlideToggleModule } from '@angular/material/slide-toggle';
    
    @Component({
      selector: 'app-root',
      imports: [MatSlideToggleModule],
      templateUrl: './app.component.html',
      styleUrl: './app.component.css'
    })
    export class AppComponent {
      title = 'my-material-app';
    }
    ```
    
    <Note>
      This example uses standalone components. If you're using NgModules, add `MatSlideToggleModule` to your module's `imports` array instead.
    </Note>
  </Step>
  
  <Step title="Use the Component">
    Add the component to your template:
    
    ```html app.component.html
    <mat-slide-toggle>Toggle me!</mat-slide-toggle>
    ```
  </Step>
  
  <Step title="Run Your App">
    Start the development server:
    
    ```bash
    ng serve
    ```
    
    Open [http://localhost:4200](http://localhost:4200) in your browser. You should see a Material Design slide toggle!
  </Step>
</Steps>

## Step 2: Build a Toolbar

Let's create a Material toolbar for your application header.

<Steps>
  <Step title="Import Toolbar and Button Modules">
    ```typescript app.component.ts
    import { Component } from '@angular/core';
    import { MatToolbarModule } from '@angular/material/toolbar';
    import { MatButtonModule } from '@angular/material/button';
    import { MatIconModule } from '@angular/material/icon';
    
    @Component({
      selector: 'app-root',
      imports: [
        MatToolbarModule,
        MatButtonModule,
        MatIconModule
      ],
      templateUrl: './app.component.html',
      styleUrl: './app.component.css'
    })
    export class AppComponent {
      title = 'My Material App';
    }
    ```
  </Step>
  
  <Step title="Create the Toolbar">
    ```html app.component.html
    <mat-toolbar color="primary">
      <button mat-icon-button>
        <mat-icon>menu</mat-icon>
      </button>
      <span>{{ title }}</span>
      <span class="spacer"></span>
      <button mat-icon-button>
        <mat-icon>account_circle</mat-icon>
      </button>
    </mat-toolbar>
    ```
  </Step>
  
  <Step title="Add Styles">
    ```css app.component.css
    .spacer {
      flex: 1 1 auto;
    }
    ```
  </Step>
</Steps>

## Step 4: Add a Form

Let's create a simple form using Material form fields.

<Steps>
  <Step title="Import Form Modules">
    ```typescript app.component.ts
    import { Component } from '@angular/core';
    import { FormControl, ReactiveFormsModule, Validators } from '@angular/forms';
    import { MatToolbarModule } from '@angular/material/toolbar';
    import { MatButtonModule } from '@angular/material/button';
    import { MatIconModule } from '@angular/material/icon';
    import { MatCardModule } from '@angular/material/card';
    import { MatFormFieldModule } from '@angular/material/form-field';
    import { MatInputModule } from '@angular/material/input';
    
    @Component({
      selector: 'app-root',
      imports: [
        ReactiveFormsModule,
        MatToolbarModule,
        MatButtonModule,
        MatIconModule,
        MatCardModule,
        MatFormFieldModule,
        MatInputModule
      ],
      templateUrl: './app.component.html',
      styleUrl: './app.component.css'
    })
    export class AppComponent {
      title = 'My Material App';
      email = new FormControl('', [Validators.required, Validators.email]);
      
      getErrorMessage() {
        if (this.email.hasError('required')) {
          return 'You must enter a value';
        }
        return this.email.hasError('email') ? 'Not a valid email' : '';
      }
    }
    ```
  </Step>
  
  <Step title="Add Form to Template">
    ```html app.component.html
    <mat-toolbar color="primary">
      <button mat-icon-button>
        <mat-icon>menu</mat-icon>
      </button>
      <span>{{ title }}</span>
      <span class="spacer"></span>
      <button mat-icon-button>
        <mat-icon>account_circle</mat-icon>
      </button>
    </mat-toolbar>
    
    <div class="content">
      <mat-card>
        <mat-card-header>
          <mat-card-title>Contact Us</mat-card-title>
          <mat-card-subtitle>Send us a message</mat-card-subtitle>
        </mat-card-header>
        <mat-card-content>
          <mat-form-field appearance="outline" class="full-width">
            <mat-label>Email</mat-label>
            <input matInput placeholder="pat@example.com" [formControl]="email" required>
            <mat-error *ngIf="email.invalid">{{getErrorMessage()}}</mat-error>
          </mat-form-field>
          
          <mat-form-field appearance="outline" class="full-width">
            <mat-label>Message</mat-label>
            <textarea matInput rows="5" placeholder="Your message"></textarea>
          </mat-form-field>
        </mat-card-content>
        <mat-card-actions>
          <button mat-raised-button color="primary">SEND</button>
          <button mat-button>CANCEL</button>
        </mat-card-actions>
      </mat-card>
    </div>
    ```
  </Step>
  
  <Step title="Add Form Styles">
    ```css app.component.css
    .spacer {
      flex: 1 1 auto;
    }
    
    .content {
      padding: 16px;
      max-width: 800px;
      margin: 0 auto;
    }
    
    mat-card {
      margin-bottom: 16px;
    }
    
    .full-width {
      width: 100%;
    }
    ```
  </Step>
</Steps>

## Common Component Patterns

<AccordionGroup>
  <Accordion title="Using Forms" icon="input-text">
    Material form fields work with both template-driven and reactive forms:
    
    ```typescript
    import { MatFormFieldModule } from '@angular/material/form-field';
    import { MatInputModule } from '@angular/material/input';
    import { ReactiveFormsModule } from '@angular/forms';
    ```
    
    All Material form components support Angular forms and validation.
  </Accordion>
  
  <Accordion title="Theming Components" icon="palette">
    Most components accept a `color` attribute for theming:
    
    ```html
    <button mat-raised-button color="primary">Primary</button>
    <button mat-raised-button color="accent">Accent</button>
    <button mat-raised-button color="warn">Warn</button>
    ```
  </Accordion>
  
  <Accordion title="Accessibility" icon="universal-access">
    Material components are accessible by default. Always provide:
    
    - Labels for form fields
    - `aria-label` for icon buttons
    - Proper heading hierarchy
    - Keyboard navigation support
  </Accordion>
  
  <Accordion title="Component Variants" icon="shapes">
    Many components have multiple variants:
    
    ```html
    <!-- Buttons -->
    <button mat-button>Basic</button>
    <button mat-raised-button>Raised</button>
    <button mat-flat-button>Flat</button>
    <button mat-stroked-button>Stroked</button>
    
    <!-- Form Fields -->
    <mat-form-field appearance="fill">...</mat-form-field>
    <mat-form-field appearance="outline">...</mat-form-field>
    ```
  </Accordion>
</AccordionGroup>

## Using Schematics

Angular Material includes schematics to generate pre-built components:

<CodeGroup>
```bash Navigation
ng generate @angular/material:navigation my-nav
```

```bash Table
ng generate @angular/material:table my-table
```

```bash Dashboard
ng generate @angular/material:dashboard my-dashboard
```

```bash Address Form
ng generate @angular/material:address-form my-form
```
</CodeGroup>

These commands generate complete, working components with Material Design.

See the [Schematics Guide](/essentials/schematics) for more details.
